cmake_minimum_required(VERSION 3.22)
project(WindowAdapterExample LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "Windows is required")
endif()
set(CMAKE_CXX_STANDARD 20)
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
if(MSVC)
    add_compile_options(/utf-8 /permissive-)
endif()
add_library(SampleCaptionAdapter SHARED sample-caption-adapter.cpp ../chrome.cpp)
target_link_libraries(SampleCaptionAdapter PRIVATE comctl32 dwmapi uxtheme)
