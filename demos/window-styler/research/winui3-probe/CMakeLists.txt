cmake_minimum_required(VERSION 3.22)
project(WinUI3ReadOnly LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_library(WinUI3ReadOnly SHARED probe.cpp probe.def)
set_target_properties(WinUI3ReadOnly PROPERTIES OUTPUT_NAME "WinUI3ReadOnly-v2")
target_include_directories(WinUI3ReadOnly PRIVATE ../../out/winui3-sdk)
target_link_libraries(WinUI3ReadOnly PRIVATE windowsapp ole32 oleaut32)
add_executable(WinUI3Research loader.cpp)
add_library(WinUI3RootsProbe SHARED native-roots.cpp)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/native-roots.cpp" roots_revision)
string(SUBSTRING "${roots_revision}" 0 12 roots_revision)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS native-roots.cpp)
set_target_properties(WinUI3RootsProbe PROPERTIES OUTPUT_NAME "WinUI3RootsProbe-${roots_revision}")
target_include_directories(WinUI3RootsProbe PRIVATE ../../out/winui3-sdk)
target_link_libraries(WinUI3RootsProbe PRIVATE windowsapp ole32 oleaut32 comctl32)
add_library(WinUI3InlineProbe SHARED inline-experiment.cpp)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/inline-experiment.cpp" inline_revision)
string(SUBSTRING "${inline_revision}" 0 12 inline_revision)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS inline-experiment.cpp)
set_target_properties(WinUI3InlineProbe PROPERTIES OUTPUT_NAME "WinUI3InlineProbe-${inline_revision}")
target_include_directories(WinUI3InlineProbe PRIVATE ../../out/winui3-sdk)
target_link_libraries(WinUI3InlineProbe PRIVATE windowsapp ole32 oleaut32 comctl32)
add_executable(WinUI3Target WIN32 target.cpp)
target_include_directories(WinUI3Target PRIVATE ../../out/winui3-sdk)
target_link_libraries(WinUI3Target PRIVATE windowsapp ole32 user32 comctl32)
