# Research configurations coexist with already pinned release bootstraps in a
# user's process. Give only this standalone project's bootstrap a distinct name
# per configuration, and keep Session's requested filename in sync. The main
# product's build and its cross-installation byte-identity guard are unchanged.
get_target_property(research_hook_name MacdowsStyleHook OUTPUT_NAME)
# Isolate bootstrap registries too: a pinned older copy otherwise resolves
# Adapter272 from its old directory even when a new research DLL was built.
file(GLOB research_revision_files CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*.h" "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp")
set(research_revision "")
foreach(source IN LISTS research_revision_files)
    file(SHA256 "${source}" part)
    string(APPEND research_revision "${part}")
endforeach()
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/../winui3-probe/title-read.cpp" title_read_source_hash)
string(APPEND research_revision "${title_read_source_hash}")
foreach(source native-island.cpp native-island-frame.h)
    set(source "${CMAKE_CURRENT_SOURCE_DIR}/../winui3-probe/${source}")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${source}")
    file(SHA256 "${source}" part)
    string(APPEND research_revision "${part}")
endforeach()
file(SHA256 "${CMAKE_CURRENT_LIST_FILE}" part)
string(SHA256 research_revision "${research_revision}${part}")
string(SUBSTRING "${research_revision}" 0 12 research_revision)
string(APPEND research_hook_name "-research-${research_revision}-$<LOWER_CASE:$<CONFIG>>")
set_target_properties(MacdowsStyleHook PROPERTIES OUTPUT_NAME "${research_hook_name}")
