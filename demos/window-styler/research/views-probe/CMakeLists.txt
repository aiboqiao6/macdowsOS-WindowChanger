cmake_minimum_required(VERSION 3.22)
project(ViewsResearch LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_compile_options($<$<COMPILE_LANGUAGE:CXX>:/utf-8>)
set(root "${CMAKE_CURRENT_SOURCE_DIR}/../..")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS probe.cpp)
add_library(minhook STATIC ${root}/third_party/minhook/src/buffer.c ${root}/third_party/minhook/src/hook.c ${root}/third_party/minhook/src/trampoline.c ${root}/third_party/minhook/src/hde/hde64.c)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/probe.cpp" revision)
string(SUBSTRING ${revision} 0 12 revision)
add_library(ViewsProbe SHARED probe.cpp)
target_compile_definitions(ViewsProbe PRIVATE VIEWS_PROBE_DLL)
target_link_libraries(ViewsProbe PRIVATE minhook)
set_target_properties(ViewsProbe PROPERTIES OUTPUT_NAME "ViewsProbe-${revision}")
add_executable(ViewsResearch probe.cpp)
target_compile_definitions(ViewsResearch PRIVATE VIEWS_PROBE_NAME=L"ViewsProbe-${revision}.dll")
add_dependencies(ViewsResearch ViewsProbe)
add_library(HookRegistrationProbe SHARED hook-registration.cpp)
